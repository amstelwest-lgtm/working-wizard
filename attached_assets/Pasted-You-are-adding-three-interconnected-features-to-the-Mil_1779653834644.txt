AI CALL FOR STEP SUGGESTIONS — IMPLEMENTATION
═══════════════════════════════════════════════════════════════

Route this through a server-side function to keep API keys 
server-side. Use Gemini Flash as the model.

const generateInterventionSuggestions = async ({
  smeContext,
  ratioContext,
  accountantPrompt
}) => {

  const systemContext = `
    SME Name: ${smeContext.name}
    Industry: ${smeContext.industry || 'not specified'}
    Location: ${smeContext.location || 'South Africa'}
    Annual Revenue: R${smeContext.revenue?.toLocaleString('en-ZA') || 'unknown'}
    
    Ratio being addressed: ${ratioContext.ratio_name}
    Current value: ${ratioContext.current_value}
    Health tier: ${ratioContext.health_tier}
    Health score: ${ratioContext.health_score}/100
    Prior period score: ${ratioContext.prior_period_score}/100
    Trend: ${ratioContext.health_score > ratioContext.prior_period_score ? 'Improving' : 'Deteriorating'}
  `;

  const prompt = `
    You are a South African SME financial advisor helping an 
    accountant identify specific intervention steps for their client.
    
    CLIENT CONTEXT:
    ${systemContext}
    
    ACCOUNTANT'S QUESTION:
    "${accountantPrompt}"
    
    Generate exactly 5 specific, practical intervention steps 
    tailored to this client's exact situation.
    
    Each step must:
    - Be specific to THIS client's numbers and context
    - Reference the actual ratio value where relevant
    - Be appropriate for a South African SME
    - Be more specific than the generic playbook steps
    - Reflect the accountant's question and context
    
    Return ONLY valid JSON. No markdown, no explanation:
    
    {
      "steps": [
        {
          "step_title": "string — 5-10 words, imperative verb, specific",
          "step_description": "string — 2-4 sentences, specific to this client",
          "timeframe": "immediate|week_1_2|month_1|month_1_3|month_3_6|year_1",
          "effort": "low|medium|high",
          "impact": "low|medium|high",
          "category": "revenue|cost|cash|structure|operations|risk|people",
          "reasoning": "string — one sentence explaining why this is relevant 
                        to this specific client's situation"
        }
      ]
    }
  `;

  const response = await fetch(
    `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${process.env.GEMINI_API_KEY}`,
    {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        contents: [{ parts: [{ text: prompt }] }],
        generationConfig: {
          temperature: 0.7,
          responseMimeType: 'application/json'
        }
      })
    }
  );

  const data = await response.json();
  const rawText = data.candidates?.[0]?.content?.parts?.[0]?.text;
  
  if (!rawText) throw new Error('No response from AI');
  
  try {
    const parsed = JSON.parse(rawText);
    
    // Save the session for audit trail
    await supabase.from('intervention_ai_sessions').insert({
      sme_id: smeContext.id,
      submission_period: smeContext.current_period,
      ratio_key: ratioContext.ratio_key,
      accountant_prompt: accountantPrompt,
      ai_response: parsed,
      model_used: 'gemini-1.5-flash',
      created_by: currentUser.id
    });
    
    return parsed.steps;
    
  } catch {
    throw new Error('Could not parse AI response');
  }
};

Use Supabase real-time subscriptions so that when the 
accountant signs off a step or adds a custom step, the 
SME owner's view updates immediately without a page refresh.

// In the SME dashboard, subscribe to changes
const setupRealtimeSubscriptions = (smeId, period) => {
  
  const signoffChannel = supabase
    .channel(`signoffs-${smeId}`)
    .on('postgres_changes', {
      event: '*',
      schema: 'public',
      table: 'intervention_signoffs',
      filter: `sme_id=eq.${smeId}`
    }, (payload) => {
      // Update the relevant step card in the UI
      updateSignoffState(payload);
    })
    .subscribe();

  const customStepsChannel = supabase
    .channel(`custom-steps-${smeId}`)
    .on('postgres_changes', {
      event: '*',
      schema: 'public',
      table: 'intervention_custom_steps',
      filter: `sme_id=eq.${smeId}`
    }, (payload) => {
      // Add, update, or remove custom step in the UI
      updateCustomSteps(payload);
    })
    .subscribe();

  return () => {
    supabase.removeChannel(signoffChannel);
    supabase.removeChannel(customStepsChannel);
  };
};

PART 7 — USAGE TRACKING FOR FUTURE BILLING
═══════════════════════════════════════════════════════════════

Track AI consultation usage per accountant firm per month.
This supports the future metered billing model.

CREATE TABLE ai_usage_tracking (
  id                uuid primary key default gen_random_uuid(),
  accountant_id     uuid references auth.users(id),
  firm_id           uuid,
  month             date,
  feature           text,
  units_used        integer default 0,
  created_at        timestamp with time zone default now(),
  UNIQUE(accountant_id, month, feature)
);

-- Increment on each AI suggestion generation
-- Use an upsert so the counter grows correctly:

await supabase.rpc('increment_ai_usage', {
  p_accountant_id: currentUser.id,
  p_feature: 'custom_step_generation',
  p_month: new Date().toISOString().slice(0, 7) + '-01'
});

-- Create the increment function in Supabase:
CREATE OR REPLACE FUNCTION increment_ai_usage(
  p_accountant_id uuid,
  p_feature text,
  p_month date
) RETURNS void AS $$
BEGIN
  INSERT INTO ai_usage_tracking 
    (accountant_id, month, feature, units_used)
  VALUES 
    (p_accountant_id, p_month, p_feature, 1)
  ON CONFLICT (accountant_id, month, feature)
  DO UPDATE SET units_used = ai_usage_tracking.units_used + 1;
END;
$$ LANGUAGE plpgsql;
